1. Հայաստանի Հանրապետության կառավարության մի շարք որոշումներում կատարել հետևյալ փոփոխությունները`
1) 2009 թվականի փետրվարի 5-ի «Պետական բյուջեից բյուջետային վարկ տրամադրելու մասին» N 162-Ա որոշման 1-ին կետում «բյուջեով նախատեսված Հայաստանի Հանրապետության կառավարության պահուստային ֆոնդի հաշվին» բառերը փոխարինել «բյուջեից» բառով, իսկ «բյուջետային ծախսերի տնտեսագիտական դասակարգման» բառերը` «բյուջետային ելքերի տնտեսագիտական դասակարգման» բառերով.
2) 2009 թվականի փետրվարի 19-ի «Հայաստանի Հանրապետության պետական բյուջեից բյուջետային վարկ տրամադրելու մասին» N 179-Ա որոշման 1-ին կետում «բյուջեով նախատեսված Հայաստանի Հանրապետության կառավարության պահուստային ֆոնդի հաշվին» բառերը փոխարինել «բյուջեից» բառով, իսկ «բյուջետային ծախսերի տնտեսագիտական դասակարգման» բառերը` «բյուջետային ելքերի տնտեսագիտական դասակարգման» բառերով.
3) 2009 թվականի փետրվարի 12-ի «Թամարա Ֆրութ» փակ բաժնետիրական ընկերության կողմից առաջարկվող օրգանիկ այգիների ստեղծման ծրագրին հավանություն տալու և Հայաստանի Հանրապետության գյուղատնտեսության նախարարությանը գումար հատկացնելու մասին» N 180-Ա որոշման 2-րդ կետում «բյուջետային ծախսերի տնտեսագիտական դասակարգման» բառերը փոխարինել «բյուջետային ելքերի տնտեսագիտական դասակարգման» բառերով.
4) 2009 թվականի մարտի 26-ի «Հայաստանի Հանրապետության ֆինանսների նախարարությանը գումար հատկացնելու մասին» N 300-Ա որոշման մեջ «հատկացնել 100,000.0 ԱՄՆ-ի դոլարին համարժեք դրամ` Հայաստանի Հանրապետության 2009 թվականի պետական բյուջեով նախատեսված Հայաստանի Հանրապետության կառավարության պահուստային ֆոնդի հաշվին» բառերը փոխարինել «Հայաստանի Հանրապետության 2009 թվականի պետական բյուջեից հատկացնել 100,000.0 ԱՄՆ-ի դոլարին համարժեք դրամ» բառերով, իսկ «բյուջետային ծախսերի տնտեսագիտական դասակարգման» բառերը` «բյուջետային ելքերի տնտեսագիտական դասակարգման» բառերով.
5) 2009 թվականի ապրիլի 2-ի «Հայաստանի Հանրապետության պետական բյուջեից բյուջետային վարկ տրամադրելու մասին» N 336-Ա որոշման 1-ին կետում «բյուջեով նախատեսված Հայաստանի Հանրապետության կառավարության պահուստային ֆոնդի հաշվին» բառերը փոխարինել «բյուջեից» բառով, իսկ «բյուջետային ծախսերի տնտեսագիտական դասակարգման» բառերը` «բյուջետային ելքերի տնտեսագիտական դասակարգման» բառերով,
6) 2009 թվականի մայիսի 21-ի «Հայաստանի Հանրապետության էկոնոմիկայի նախարարությանը գումար հատկացնելու մասին» N 563-Ա որոշման 1-ին կետում «Հայաստանի Հանրապետության 2009 թվականի պետական բյուջեով նախատեսված Հայաստանի Հանրապետության կառավարության պահուստային ֆոնդի հաշվին (բյուջետային ծախսերի տնտեսագիտական դասակարգման «Վարկերի և փոխառությունների հիմնական գումարների մարում» հոդվածով)» բառերը փոխարինել «Հայաստանի Հանրապետության 2009 թվականի պետական բյուջեից (բյուջետային ելքերի տնտեսագիտական դասակարգման «Վարկերի և փոխատվությունների տրամադրում» հոդվածով)» բառերով:
