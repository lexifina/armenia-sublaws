ա) արձանագրության քարտուղարության հետ և կողմերի միջև կենսաբանական անվտանգության վերաբերյալ տեղեկատվության փոխանակումն իրականացնում է Հայաստանի Հանրապետության բնապահպանության նախարարությունը` իրազեկելով Հայաստանի Հանրապետության առողջապահության նախարարությանը, Հայաստանի Հանրապետության առևտրի և տնտեսական զարգացման նախարարությանն ու Հայաստանի Հանրապետության գյուղատնտեսության նախարարությանը.
