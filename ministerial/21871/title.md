ՄԱՐԶՊԵՏԱՐԱՆԻ (ԵՐԵՎԱՆԻ ՔԱՂԱՔԱՊԵՏԱՐԱՆԻ) ԵՐԵԽԱՆԵՐԻ ԻՐԱՎՈՒՆՔՆԵՐԻ ՊԱՇՏՊԱՆՈՒԹՅԱՆ ԲԱԺՆԻ I. ԸՆԴՀԱՆՈՒՐ ԴՐՈՒՅԹՆԵՐ 1. Հայաստանի Հանրապետության .................................................. մարզպետարանի (այսուհետ` մարզպետարան), Երևանի քաղաքապետարանի (այսուհետ` քաղաքապետարան) աշխատակազմի երեխաների իրավունքների պաշտպանության բաժինը (այսուհետ` բաժին) մարզպետարանի (քաղաքապետարանի) աշխատակազմի կառուցվածքային ստորաբաժանում է: 2. Բաժինը ստեղծվում, վերակազմակերպվում և դրա գործունեությունը դադարեցվում է Հայաստանի Հանրապետության կառավարության որոշմամբ: Բաժինը գործում է իր կանոնադրության համաձայն, որը հաստատում է Հայաստանի Հանրապետության .......................... մարզպետը (այսուհետ` մարզպետ), Երևանի քաղաքապետը (այսուհետ` քաղաքապետ): 3. Բաժինն իր գործունեության ընթացքում ղեկավարվում է Հայաստանի Հանրապետության Սահմանադրությամբ, Հայաստանի Հանրապետության օրենքներով, Հայաստանի Հանրապետության Նախագահի հրամանագրերով, Հայաստանի Հանրապետության կառավարության և Հայաստանի Հանրապետության վարչապետի որոշումներով, այլ իրավական ակտերով և սույն կանոնադրությամբ, ինչպես նաև մարզպետի (քաղաքապետի) որոշումներով, մարզպետարանի (քաղաքապետարանի) աշխատակազմի ղեկավարի հրամաններով, ցուցումներով, մարզպետի (քաղաքապետի) համապատասխան տեղակալի ցուցումներով: 4. Բաժինը մարզպետի (քաղաքապետի), նրա համապատասխան տեղակալի և մարզպետարանի (քաղաքապետարանի) աշխատակազմի ղեկավարի հանձնարարությամբ, իր իրավասությունների շրջանակներում, համագործակցում է մարզի խնամակալության և հոգաբարձության մարմինների, սոցիալական ծառայության տարածքային գործակալությունների, զբաղվածության տարածքային կենտրոնների, բժշկասոցիալական փորձաքննական հանձնաժողովների, Հայաստանի Հանրապետության կառավարությանն առընթեր Հայաստանի Հանրապետության ոստիկանության (այսուհետ` ոստիկանություն) անչափահասների գործերով զբաղվող ծառայությունների, հասարակական կազմակերպությունների, երեխաների խնամքի և պաշտպանության այլ հաստատությունների հետ: 5. Բաժինը իր գործունեությունը կազմակերպում և իրականացնում է աշխատանքային ծրագրերով: Բաժնի աշխատանքային ծրագիրը յուրաքանչյուր տարի հաստատում է մարզպետը (քաղաքապետը): II. ԲԱԺՆԻ ԽՆԴԻՐՆԵՐԸ 6. Բաժնի խնդիրներն են` ա) մարզում (Երևան քաղաքում) առանց ծնողական խնամքի մնացած երեխաների իրավունքների պաշտպանության ապահովումը, բ) մարզում (Երևան քաղաքում) ընտանիքի, կանանց և երեխաների պաշտպանությանն ուղղված քաղաքականության իրականացման ապահովումը, գ) մարզում (Երևան քաղաքում) ընտանիքի, կանանց և երեխաների սոցիալական պաշտպանությանն ուղղված պետական նպատակային ծրագրերի կատարման ապահովումը: III. Բաժինն իրականացնում է հետևյալ գործառույթները` ա) իրականացնում է ընտանիքի, կանանց և երեխաների պաշտպանությանն ուղղված քաղաքականությունը մարզերում, բ) մշակում և իրականացնում է ընտանիքի, կանանց և երեխաների պաշտպանությանն ուղղված մարզային (քաղաքային) ծրագրեր, գ) մշակում և իրականացնում է կյանքի դժվարին իրավիճակներում գտնվող երեխաների սոցիալական պաշտպանությանն ու հասցեական օգնությանն ուղղված ծրագրեր, դ) ուսումնասիրման և գնահատման արդյունքների հիման վրա կազմում է առանց ծնողական խնամքի մնացած երեխաների և առանց ծնողական խնամքի մնացած երեխաների թվին պատկանող անձանց սոցիալ-հոգեբանական վերականգնողական ծրագրեր, ե) իրականացնում է հանրակրթական դպրոցից դուրս մնացած և կրթության առանձնահատուկ պայմանների կարիք ունեցող երեխաների կրթական իրավունքի պաշտպանությունը, զ) ոստիկանության անչափահասների գործերով զբաղվող ծառայությունների հետ համատեղ քննարկում է մուրացիկ, թափառաշրջիկ, իրավախախտ, հանցագործություն կատարած անչափահասների տեղավորման հետ կապված կողմնորոշման խնդիրները, խնամատար ընտանիք տեղավորելու մասին, ժ) վկայագրում է երեխայի կրթության առանձնահատուկ պայմանների կարիքը և տալիս է երեխայի ծնողին (կամ խնամակալին) եզրակացություն հատուկ ուսումնական հաստատության պատկանող անձանց տարածքային հաշվառման մատյանները և ձևավորում է տվյալ մարզի (Երևան քաղաքի) տարածքում բնակվող առանց ծնողական խնամքի մնացած երեխաների տվյալների տարածքային շտեմարան և առանց ծնողական խնամքի մնացած երեխաների թվին պատկանող անձանց տվյալների տարածքային շտեմարան, ժբ) առանց ծնողական խնամքի մնացած երեխաների և առանց ծնողական խնամքի մնացած երեխաների թվին պատկանող անձանց լուսանկարները և անձնական քարտերի պատճենները ուղարկում է Հայաստանի Հանրապետության աշխատանքի և սոցիալական հարցերի նախարարություն` երեխաներին կենտրոնացված հաշվառման վերցնելու և կազմակերպում է Հայաստանի Հանրապետության տարածքում մշտապես բնակվող Հայաստանի Հանրապետության քաղաքացիների ընտանիքներում դաստիարակության հանձնելուն աջակցելը, ինչպես նաև առանց ծնողական խնամքի մնացած երեխաների թվին պատկանող անձանց Հայաստանի Հանրապետության օրենսդրությամբ սահմանված սոցիալական երաշխիքները ապահովելու համար, ժգ) տարածքային հաշվառման վերցված առանց ծնողական խնամքի մնացած երեխայի կարգավիճակի իրավական հիմքերի փոփոխման դեպքում (ծնողի/ների ծնողական իրավունքների վերականգնման, երեխայի որդեգրման և այլն) 7-օրյա ժամկետում այդ մասին տեղեկացնում է Հայաստանի Հանրապետության աշխատանքի և սոցիալական հարցերի նախարարությանը, ժդ) որդեգրել ցանկացող Հայաստանի Հանրապետության տարածքում բնակվող Հայաստանի Հանրապետության քաղաքացուց ստացված բոլոր անհրաժեշտ փաստաթղթերի և անձի ու նրա կենսապայմանների ուսումնասիրության ակտի հիման վրա որդեգրել ցանկացող անձին տալիս է որդեգրման հնարավորության մասին եզրակացություն, ժե) դրական եզրակացության հիման վրա որդեգրել ցանկացող անձին հաշվառում է որպես որդեգրողի թեկնածու, տվյալները գրանցում է հաշվառման մատյանում, լրացնում որդեգրողի թեկնածուի անձնական քարտը և հաշվառման պահից եռօրյա ժամկետում պատճենը փոխանցում է Հայաստանի Հանրապետության աշխատանքի և սոցիալական հարցերի նախարարություն, ժզ) մարզպետարանի (քաղաքապետարանի)` որդեգրման հնարավորության բացասական եզրակացության մասին հնգօրյա ժամկետում գրավոր տեղեկացնում է դիմողին, ժէ) խնամատար ծնող դառնալու ցանկություն հայտնած անձից բոլոր անհրաժեշտ փաստաթղթերը ստանալուց հետո մեկամսյա ժամկետում կատարում է այդ անձի ու նրա կենսապայմանների ուսումնասիրություն և կազմում է համապատասխան ակտ, ժը) խնամատար ծնող դառնալու ցանկություն հայտնած անձից ստացված բոլոր անհրաժեշտ փաստաթղթերի և անձի ու նրա կենսապայմանների ուսումնասիրության ակտի հիման վրա խնամատար ծնող դառնալու ցանկություն հայտնած անձին տալիս է խնամատար ծնող դառնալու հնարավորության մասին եզրակացություն, ժթ) դրական եզրակացության հիման վրա խնամատար ծնող դառնալու ցանկություն հայտնած անձին հաշվառում է որպես խնամատար ծնողի թեկնածու, տվյալները գրանցում է հաշվառման մատյանում և լրացնում խնամատար ծնողի թեկնածուի անձնական քարտը և հաշվառման պահից եռօրյա ժամկետում պատճենը փոխանցում է Հայաստանի Հանրապետության աշխատանքի և սոցիալական հարցերի նախարարություն, է) իրականացնում է որդեգրման ենթակա երեխաների հաշվառումը և եռօրյա ժամկետում անձնական քարտի պատճենները ներկայացնում է Հայաստանի Հանրապետության աշխատանքի և սոցիալական հարցերի նախարարություն` կենտրոնացած հաշվառման վերցնելու համար, ը) իրականացնում է Հայաստանի Հանրապետության օրենսդրությամբ իրեն վերապահված այլ լիազորություններ: IV. ԲԱԺՆԻ ՊԵՏԸ ԵՎ ԱՇԽԱՏՈՂՆԵՐԸ 8. Բաժնի պետը պաշտոնի է նշանակվում և պաշտոնից ազատվում է «Քաղաքացիական ծառայության մասին» Հայաստանի Հանրապետության օրենքով սահմանված կարգով: 9. Բաժնի պետը հաշվետու է մարզպետին (քաղաքապետին), մարզպետի (քաղաքապետի)` բաժնի գործունեությունը համակարգող համապատասխան տեղակալին, «Քաղաքացիական ծառայության մասին» Հայաստանի Հանրապետության օրենքով նախատեսված դեպքերում` մարզպետարանի (քաղաքապետարանի) աշխատակազմի ղեկավարին: 10. Բաժնի պետը` ա) կազմակերպում, ծրագրում, համակարգում, ղեկավարում և վերահսկում է բաժնի ընթացիկ գործունեությունը, բ) առաջարկություններ է ներկայացնում օրենքով, այլ իրավական ակտերով և սույն կանոնադրությամբ նախատեսված գործունեության բնագավառի նպատակների և խնդիրների վերաբերյալ մարզպետին (քաղաքապետին) կամ նրա համապատասխան տեղակալին, գ) ատեստավորումից առնվազն երկու շաբաթ առաջ ներկայացնում է իր անմիջական ենթակայության տակ գտնվող քաղաքացիական ծառայողների ծառայողական բնութագրերը, դ) իր լիազորությունների շրջանակներում ապահովում է բաժնի և այլ մարմինների փոխհարաբերությունները, ե) ստորագրում է բաժնի անունից պատրաստվող փաստաթղթերը, զ) բաժնի լիազորությունների շրջանակներում նախապատրաստում է առաջարկություններ, տեղեկանքներ, հաշվետվություններ, միջնորդագրեր, զեկուցագրեր և այլ գրություններ, է) մարզպետարանի (քաղաքապետարանի) աշխատակազմի ղեկավարի հանձնարարությամբ մասնակցում է պետական և այլ մարմինների խորհրդակցություններին, ը) իրականացնում է Հայաստանի Հանրապետության օրենսդրությամբ, ինչպես նաև իր պաշտոնի անձնագրով սահմանված այլ լիազորություններ, թ) բաժնի պետի բացակայության կամ պաշտոնական պարտականությունների կատարման ժամանակավոր անհնարինության դեպքում բաժնի պետի պարտականությունների կատարման հարցը կարգավորվում է «Քաղաքացիական ծառայության մասին» Հայաստանի Հանրապետության օրենքի 18-րդ հոդվածի պահանջներին համապատասխան, ժ) բաժնի աշխատողներն աշխատանքի են ընդունվում և աշխատանքից ազատվում «Քաղաքացիական ծառայության մասին» Հայաստանի Հանրապետության օրենքով սահմանված կարգով, ժա) բաժնի աշխատողներն իրականացնում են Հայաստանի Հանրապետության օրենսդրությամբ, ինչպես նաև իրենց պաշտոնների անձնագրերով նախատեսված լիազորությունները, ապահովում են իրենց անմիջական ղեկավարների հանձնարարականների կատարումը: V. ԲԱԺՆԻ ՔԱՂԱՔԱՑԻԱԿԱՆ ԾԱՌԱՅՈՂՆԵՐԻ ՄԱՍՆԱԳԻՏԱԿԱՆ ԳԻՏԵԼԻՔՆԵՐԸ ԵՎ ԱՇԽԱՏԱՆՔԱՅԻՆ ՈՒՆԱԿՈՒԹՅՈՒՆՆԵՐԸ 11. Բաժնում պետք է աշխատեն իրավագիտություն, հոգեբանություն, սոցիալական աշխատանք, բժիշկ, մանկավարժ մասնագիտությամբ առնվազն մեկական քաղաքացիական ծառայողներ: 12. Բաժնի պետը, գլխավոր մասնագետը, ինչպես նաև քաղաքացիական ծառայության առաջատար պաշտոնների խմբում ընդգրկված պաշտոններում աշխատող քաղաքացիական ծառայողները` ա) ունեն իրենց լիազորությունների հետ կապված ընտանիքի, կանանց և երեխաների հիմնահարցերի ոլորտի, որդեգրման և խնամատար ընտանիքի իրավական ակտերի իմացություն, բ) ունեն համակարգչով և ժամանակակից այլ տեխնիկական միջոցներով աշխատելու ունակություն, ինչպես նաև տարբեր տեղեկատվական շտեմարանների համակարգչային ծրագրերի շահագործման համար պահանջվող գիտելիքներ: 13. Քաղաքացիական ծառայության կրտսեր պաշտոնների խմբում ընդգրկված պաշտոններում աշխատող քաղաքացիական ծառայողները` ա) ունեն իրենց լիազորություների հետ կապված (մասնավորապես երեխաների իրավունքների, ընտանիքի, կանանց հիմնահարցերի ոլորտի, որդեգրման և խնամատար ընտանիքի) իրավական ակտերի իմացություն, բ) ունեն համակարգչով և ժամանակակից այլ տեխնիկական միջոցներով աշխատելու ունակություն: VI. ԲԱԺՆԻ ՔԱՂԱՔԱՑԻԱԿԱՆ ԾԱՌԱՅՈՂՆԵՐԻ ԻՐԱՎՈՒՆՔՆԵՐԸ ԵՎ ՊԱՐՏԱԿԱՆՈՒԹՅՈՒՆՆԵՐԸ 14. Բաժնի պետը իրավունք ունի` ա) իր լիազորությունների շրջանակներում համագործակցել մարզպետարանի համապատասխան ստորաբաժանումների, տեղական ինքնակառավարման մարմինների, մարզում գործող հանրակրթական դպրոցների և երեխաների խնամք և պաշտպանություն իրականացնող հաստատությունների, հասարակական կազմակերպությունների հետ, բ) իր լիազորությունների շրջանակներում ստանալ անհրաժեշտ տեղեկատվություն և նյութեր այլ մարմիններից, պաշտոնատար անձանցից, գ) մասնակցել երեխաների պաշտպանության թեմաներով սեմինարների, գիտաժողովների, դ) ստուգել խնամակալության և հոգաբարձության մարմիններից ստացված տեղեկատվության ճշտությունը առանց ծնողական խնամքի մնացած երեխաների մասին: 15. Գլխավոր մասնագետը իրավունք ունի` ա) իր լիազորությունների շրջանակներում, բաժնի պետի հանձնարարությամբ, համագործակցել մարզպետարանի համապատասխան ստորաբաժանումների, տեղական ինքնակառավարման մարմինների, մարզում գործող հանրակրթական դպրոցների և երեխաների խնամք և պաշտպանություն իրականացնող հաստատությունների հետ, բ) կատարել տուն-այցեր առանց ծնողական խնամքի մնացած երեխաների ուսումնասիրման նպատակով, գ) մասնակցել երեխաների պաշտպանության թեմաներով սեմինարների, գիտաժողովների: 16. Բաժնի քաղաքացիական ծառայողը, որի պաշտոնը ընդգրկվում է առաջատար պաշտոնների խմբում, իրավունք ունի` ա) բաժնի պետի հանձնարարությամբ կատարել տնային այցելություններ որդեգրել ցանկացող անձանց, նրանց ընտանիքի անդամներին, երեխային (երեխաներին), խնամատար ընտանիքի անդամներին, անհրաժեշտության դեպքում` նրանց կենսապայմաններին ծանոթանալու նպատակով, բ) բաժնի պետի հանձնարարությամբ ստուգել Հայաստանի Հանրապետության քաղաքացի երեխա որդեգրելու համար դիմած Հայաստանի Հանրապետության քաղաքացիների կողմից ներկայացված փաստաթղթերի ճշտությունը, 17. Բաժնի պետը պարտավոր է` ա) ապահովել բաժնի առջև դրված խնդիրների լուծումը և գործառույթների իրականացումը, բ) ներկայացնել առաջարկություն մարզպետին (քաղաքապետին) կամ նրա համապատասխան տեղակալին` ընտանիքի, կանանց և երեխաների սոցիալական պաշտպանությանն ու հասցեական օգնությանն ուղղված մարզային և տեղական նշանակության ծրագրերի ձևավորման և դրանց կատարելագործման վերաբերյալ, գ) ներկայացնել Հայաստանի Հանրապետության տարածքում բնակվող Հայաստանի Հանրապետության քաղաքացի որդեգրել ցանկացող անձի՝ որդեգրման հնարավորության մասին եզրակացություն, դ) ներկայացնել Հայաստանի Հանրապետության տարածքում բնակվող Հայաստանի Հանրապետության քաղաքացի խնամատար ծնող դառնալու ցանկություն հայտնած անձին խնամատար ծնող դառնալու հնարավորության մասին եզրակացություն, ե) առանց ծնողական խնամքի մնացած երեխաների ուսումնասիրման և կարիքների գնահատման հիման վրա խնամակալության և հոգաբարձության մարմին է ներկայացնում եզրակացություն առանց ծնողական խնամքի մնացած երեխաներին մանկատուն կամ երեխաների գիշերօթիկ խնամքի հաստատություն կամ խնամատար ընտանիք տեղավորելու մասին, զ) վկայագրել երեխայի կրթության առանձնահատուկ պայմանների կարիքը և տալ երեխայի ծնողին (կամ խնամակալին) եզրակացություն հատուկ ուսումնական հաստատության ընտրության վերաբերյալ, է) ոստիկանության անչափահասների գործերով զբաղվող ծառայությունների հետ համատեղ քննարկել մուրացիկ, թափառաշրջիկ, իրավախախտ, հանցագործություն կատարած անչափահասների տեղավորման հետ կապված կողմնորոշման խնդիրները, պաշտպանել անչափահասների իրավունքներն ու շահերը, ը) ապահովել մարզում (Երևան քաղաքում) առանց ծնողական խնամքի մնացած երեխաների իրավունքների պաշտպանությունը, թ) ապահովել մարզում (Երևան քաղաքում) ընտանիքի, կանանց և երեխաների պաշտպանությանն ուղղված քաղաքականության իրականացումը, ժ) ապահովել մարզում (Երևան քաղաքում) ընտանիքի, կանանց և երեխաների սոցիալական պաշտպանությանն ուղղված պետական նպատակային ծրագրերի մշակումը և դրանց կատարումը, ժա) ապահովել հանրակրթական դպրոցից դուրս մնացած և կրթության առանձնահատուկ պայմանների կարիք ունեցող երեխաների կրթական իրավունքի պաշտպանությունը, ժբ) ապահովել այն երեխաների հաշվառումը, որոնց որդեգրման համար օրենքով սահմանված կարգով ծագել են իրավական հիմքեր, ժգ) ապահովել երեխա որդեգրելու համար դիմած Հայաստանի Հանրապետության քաղաքացիների հաշվառումը, ժդ) քաղաքացուց բոլոր անհրաժեշտ փաստաթղթերը ստանալուց հետո մեկամսյա ժամկետում կազմակերպել այդ անձի ու նրա կենսապայմանների ուսումնասիրություն և կազմել համապատասխան ակտ, ժե) ներկայացնել մարզպետին (քաղաքապետին) երեխայի իրավունքների պաշտպանության գործունեության մասին տարեկան հաշվետվություն, ժզ) կազմակերպել և համակարգել բաժնի մյուս մասնագետների աշխատանքը: 18.
